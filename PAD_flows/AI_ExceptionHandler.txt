function Normalize-Field {
    param([object]$Value)
    if ($null -eq $Value) {
        return \"\"
    }

    return ([string]$Value -replace \"`r`n\", \" \" -replace \"`r\", \" \" -replace \"`n\", \" \" -replace \"`t\", \" \").Trim()
